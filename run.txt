# Run instructions
#
# macOS/Linux (current repository)
# --------------------------------
# From the repository root:
#   cd "/Users/prokash/DWCT/Github/Prokash21.github.io"
#
# Install Hugo Extended once if Hugo is not already installed:
#   brew install hugo
#
# Start the local development server:
#   hugo server -D -s .
#
# Open the URL printed by Hugo, usually http://localhost:1313/.
# Stop the server with Ctrl+C.
#
# Build the GitHub Pages output into docs/:
#   hugo --gc --minify -s . -d docs
#
# Windows/PowerShell
# ------------------
# Use the provided PowerShell starter script `run.ps1` which will:
#  - install Hugo to .\tools once (if missing)
#  - pick a free port starting at 1313 if 1313 is already in use
#  - start the Hugo dev server and print the URL
#
# Recommended:
#  Rename this file to a script once and run it:
#    Rename-Item .\run.txt .\run.ps1
#    powershell -ExecutionPolicy Bypass -File .\run.ps1
#
